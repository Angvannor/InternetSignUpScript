# ---------------------------------------------------------------------------
# 校园网自动登录工具 —— 第三方依赖（全部可选，见下方说明）
# 安装：  python -m pip install -r requirements.txt
# ---------------------------------------------------------------------------

# 用 Edge 操作网页元素完成登录（需求文档指定方案）。
# Selenium 4 自带 Selenium Manager，不需要你手动下载 EdgeDriver。
selenium>=4.6

# 把密码存进 Windows 凭据管理器（比存本地文件更安全）。
keyring>=24.0

# keyring 在 Windows 上访问凭据管理器用到的底层库（需求文档要求列出）。
pywin32-ctypes>=0.2.2

# ---------------------------------------------------------------------------
# 重要说明：
#
# 上面三个包都是**可选**的。本工具在"什么都没装"的纯净 Python 环境下也能
# 完成登录，原因是：
#
#   校园网没登录的时候本来就上不了外网，pip 也就装不了任何东西 ——
#   这是个死循环。所以 login.py 内置了一个纯标准库的 HTTP 引擎
#   （engine_http.py），并用 Windows 自带的 DPAPI 加密保存密码。
#
# 默认的 engine = "auto" 会先用 HTTP 引擎；只有它失败时才回退到
# Selenium + Edge。装了 selenium 就能享受"真实浏览器操作"这条路径。
# ---------------------------------------------------------------------------
